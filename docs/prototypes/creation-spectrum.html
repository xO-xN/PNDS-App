// 水纹乐器 · 单次触发
SynthDef(\ripple, {
    |out = 0, freq = 110, amp = 0.2|
    var env, tone;

    env = EnvGen.kr(
        Env.perc(0.01, 1.4),
        doneAction: 2
    );
    tone = SinOsc.ar(freq);
    tone = tone + (SinOsc.ar(freq * 2) * 0.3);
    Out.ar(out, Pan2.ar(tone * env * amp));
}).add;
